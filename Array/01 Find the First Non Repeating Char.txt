// Find the first non-repeating char in the array 

#include<bits/stdc++.h>
using namespace std;

int main() {
    string name = "swisswi";
    int n = name.length();
    
    vector<int> arr(26, 0);
    
    for(int i=0;i<n;i++){
        if(arr[name[i]- 'a'] == 0  && arr[name[i]-'a'] != -1) {
            arr[name[i]-'a'] = 1;
        } else if(arr[name[i] - 'a'] == 1) {
            arr[name[i] - 'a'] = -1;
        }
    }
    
    for(int i=0;i<n;i++) {
        if(arr[name[i] - 'a'] == 1) {
            cout << name[i];
            break;
        }
    }
    
    return 0;
}
